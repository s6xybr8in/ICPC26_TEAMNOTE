\section{트러블슈팅}
\begingroup
\footnotesize
\setlist[itemize]{nosep,leftmargin=*}

\textbf{제출 전}
\begin{itemize}
  \item 예제 외에 최소 입력, 경계값, 반례를 직접 확인했는가?
  \item 최대 입력에서 시간과 메모리가 충분한가?
  \item 합, 곱, 거리, INF 계산에서 오버플로가 없는가?
  \item 제출 파일과 언어를 확인하고 디버그 출력을 지웠는가?
\end{itemize}

\textbf{틀렸습니다 (WA)}
\begin{itemize}
  \item 문제를 다시 읽고 입력 조건과 출력 형식을 확인한다.
  \item 테스트 케이스마다 배열, 그래프, 전역 상태를 초기화했는가?
  \item 0/1-based 인덱스와 닫힌 구간 $[l,r]$를 확인한다.
  \item 빈 구간, 원소 하나, 중복, 음수, 연결되지 않은 그래프를 확인한다.
  \item 초기값, 자료형, 비교 연산의 등호, 정수 나눗셈을 확인한다.
  \item 곱하기 전에 \texttt{1LL} 또는 \texttt{\_\_int128}로 승격했는가?
  \item 정렬 비교 함수가 엄격한 순서를 만족하는가? 실수 오차 처리는 적절한가?
  \item 작은 입력을 손으로 추적하거나 완전탐색 풀이와 비교한다.
  \item STL 함수와 복사한 코드의 전제조건을 다시 확인한다.
  \item 팀원에게 풀이를 설명하고 코드 검토를 부탁한다.
\end{itemize}

\textbf{런타임 에러 (RE)}
\begin{itemize}
  \item 배열 범위, 초기화하지 않은 값, 널 포인터를 확인한다.
  \item 0으로 나누기, 빈 컨테이너 접근, 잘못된 시프트가 있는가?
  \item 재귀가 종료되는가? 재귀 깊이 때문에 스택이 넘치지 않는가?
  \item 컨테이너 변경 후 무효화된 반복자나 참조를 쓰지 않는가?
  \item 로컬에서 assert, 디버그 모드, sanitizer로 재현한다.
\end{itemize}

\textbf{시간 초과 (TLE)}
\begin{itemize}
  \item 무한 루프와 잘못된 종료 조건을 먼저 확인한다.
  \item 최악의 입력에서 실제 시간복잡도를 다시 계산한다.
  \item 값 전달, 컨테이너 복사, 반복 할당, 불필요한 정렬이 있는가?
  \item 입출력 병목과 \texttt{endl}의 반복 flush를 확인한다.
  \item unordered 컨테이너의 해시 충돌과 재해시를 확인한다.
  \item 최대 크기 입력을 만들고 병목을 측정한다.
\end{itemize}

\textbf{메모리 초과 (MLE)}
\begin{itemize}
  \item 배열 크기뿐 아니라 간선, 노드, 컨테이너 부가 비용도 계산한다.
  \item 테스트 케이스 사이에 불필요한 데이터가 누적되는가?
  \item 재귀 스택, 큰 지역 배열, 복사본, 영속 노드 수를 확인한다.
\end{itemize}
\endgroup
